    tickers,

        "Test Statistic": score,
        "P-Value": p_value

position = pd.Series(0, index=z_score.index)

current_position = 0

for date in z_score.index:
    z = z_score.loc[date]

    # Enter a long position
    if current_position == 0 and z < -2:
        current_position = 1

    # Enter a short position
    elif current_position == 0 and z > 2:
        current_position = -1

    # Exit the trade when z-score returns close to zero
    elif current_position != 0 and abs(z) < 0.5:
        current_position = 0

    position.loc[date] = current_position

print(position.value_counts())
